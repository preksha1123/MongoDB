package com.example;

import com.mongodb.client.MongoClient;
import com.mongodb.client.MongoClients;
import com.mongodb.client.MongoDatabase;
import com.mongodb.client.MongoCollection;
import org.bson.Document;

public class App {
    public static void main(String[] args) {

        String uri = System.getenv("MONGODB_URI");

        if (uri == null || uri.isBlank()) {
            System.out.println("Please set MONGODB_URI.");
            return;
        }

        try (MongoClient client = MongoClients.create(uri)) {

            // Connect to the database
            MongoDatabase database =
                    client.getDatabase("collegeDB");

            // Select the students collection
            MongoCollection<Document> students =
                    database.getCollection("students");

            // Create one student document
            Document student = new Document("name", "Priya")
                    .append("age", 22)
                    .append("course", "Java");

            // Insert the document
            students.insertOne(student);

            System.out.println("Student inserted successfully!");

        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
        }
    }
}

package com.example;

import com.mongodb.client.MongoClient;
import com.mongodb.client.MongoClients;
import com.mongodb.client.MongoDatabase;
import com.mongodb.client.MongoCollection;
import org.bson.Document;

import java.util.Arrays;
import java.util.List;

public class App {
    public static void main(String[] args) {

        String uri = System.getenv("MONGODB_URI");

        if (uri == null || uri.isBlank()) {
            System.out.println("Please set MONGODB_URI.");
            return;
        }

        try (MongoClient client = MongoClients.create(uri)) {

            // Connect to database
            MongoDatabase database =
                    client.getDatabase("collegeDB");

            // Select collection
            MongoCollection<Document> students =
                    database.getCollection("students");

            // Create multiple student documents
            List<Document> studentList = Arrays.asList(

                new Document("name", "Rahul")
                    .append("age", 21)
                    .append("course", "Java"),

                new Document("name", "Neha")
                    .append("age", 20)
                    .append("course", "Python"),

                new Document("name", "Aman")
                    .append("age", 22)
                    .append("course", "MongoDB"),

                new Document("name", "Priya")
                    .append("age", 19)
                    .append("course", "C++")
            );

            // Insert all documents at once
            students.insertMany(studentList);

            System.out.println("All students inserted successfully!");

            // Display all students
            for (Document student : students.find()) {
                System.out.println(student.toJson());
            }

        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
        }
    }
}

package com.example;

import com.mongodb.client.MongoClient;
import com.mongodb.client.MongoClients;
import com.mongodb.client.MongoDatabase;
import com.mongodb.client.MongoCollection;
import com.mongodb.client.result.UpdateResult;
import org.bson.Document;

public class App {
    public static void main(String[] args) {

        String uri = System.getenv("MONGODB_URI");

        if (uri == null || uri.isBlank()) {
            System.out.println("Please set MONGODB_URI.");
            return;
        }

        try (MongoClient client = MongoClients.create(uri)) {

            MongoDatabase database =
                    client.getDatabase("collegeDB");

            MongoCollection<Document> students =
                    database.getCollection("students");

            // Find Rahul and update his age
            Document filter =
                    new Document("name", "Rahul");

            Document update =
                    new Document("$set",
                            new Document("age", 23));

            UpdateResult result =
                    students.updateOne(filter, update);

            if (result.getMatchedCount() > 0) {
                System.out.println("Student found!");
                System.out.println(
                        "Documents modified: "
                        + result.getModifiedCount());
            } else {
                System.out.println("Rahul was not found.");
            }

            // Display Rahul's updated details
            System.out.println("Updated student:");

            for (Document s : students.find(filter)) {
                System.out.println(s.toJson());
            }

        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
        }
    }
}
