package com.example;

import com.mongodb.client.MongoClient;
import com.mongodb.client.MongoClients;
import com.mongodb.client.MongoCollection;
import com.mongodb.client.MongoDatabase;
import com.mongodb.client.model.Accumulators;
import com.mongodb.client.model.Aggregates;
import com.mongodb.client.model.Filters;
import com.mongodb.client.model.Sorts;
import com.mongodb.client.model.UpdateOptions;
import com.mongodb.client.result.DeleteResult;
import com.mongodb.client.result.UpdateResult;
import org.bson.Document;

import java.util.Arrays;
import java.util.Scanner;

public class App {
    private static final Scanner sc = new Scanner(System.in);
    private static MongoCollection<Document> students;

    public static void main(String[] args) {
        String uri = System.getenv("MONGODB_URI");

        if (uri == null || uri.isBlank()) {
            System.out.println("Please set MONGODB_URI first.");
            return;
        }

        try (MongoClient client = MongoClients.create(uri)) {
            MongoDatabase database = client.getDatabase("college");
            students = database.getCollection("students");

            database.runCommand(new Document("ping", 1));
            System.out.println("MongoDB connected successfully!");

            while (true) {
                System.out.println("\n===== STUDENT MANAGEMENT =====");
                System.out.println("1. Update CGPA");
                System.out.println("2. Update Email");
                System.out.println("3. Add Skill");
                System.out.println("4. Update Phone");
                System.out.println("5. Upsert Student");
                System.out.println("6. Delete Student");
                System.out.println("7. Delete Multiple Students");
                System.out.println("8. Department Statistics");
                System.out.println("9. Average CGPA");
                System.out.println("10. Top Students");
                System.out.println("11. View Students");
                System.out.println("12. Exit");

                int choice = readInt("Enter choice: ");

                switch (choice) {
                    case 1 -> updateCGPA();
                    case 2 -> updateEmail();
                    case 3 -> addSkill();
                    case 4 -> updatePhone();
                    case 5 -> upsertStudent();
                    case 6 -> deleteStudent();
                    case 7 -> deleteMultiple();
                    case 8 -> departmentStatistics();
                    case 9 -> averageCGPA();
                    case 10 -> topStudents();
                    case 11 -> viewStudents();
                    case 12 -> {
                        System.out.println("Goodbye!");
                        return;
                    }
                    default -> System.out.println("Invalid choice!");
                }
            }
        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
        }
    }

    private static String readText(String prompt) {
        System.out.print(prompt);
        return sc.nextLine().trim();
    }

    private static int readInt(String prompt) {
        while (true) {
            try {
                return Integer.parseInt(readText(prompt));
            } catch (NumberFormatException e) {
                System.out.println("Please enter a whole number.");
            }
        }
    }

    private static double readDouble(String prompt) {
        while (true) {
            try {
                double value = Double.parseDouble(readText(prompt));
                if (Double.isFinite(value)) {
                    return value;
                }
            } catch (NumberFormatException ignored) {
                // Prompt again below.
            }
            System.out.println("Please enter a valid number.");
        }
    }

    private static Document studentFilter(String name) {
        return new Document("name", name);
    }

    private static boolean validCgpa(double cgpa) {
        if (cgpa < 0 || cgpa > 10) {
            System.out.println("CGPA must be between 0 and 10.");
            return false;
        }
        return true;
    }

    // 1. Update CGPA
    private static void updateCGPA() {
        String name = readText("Student name: ");
        double cgpa = readDouble("New CGPA (0-10): ");
        if (!validCgpa(cgpa)) return;

        UpdateResult result = students.updateOne(
                studentFilter(name),
                new Document("$set", new Document("cgpa", cgpa))
        );
        printUpdateResult(result);
    }

    // 2. Update Email
    private static void updateEmail() {
        String name = readText("Student name: ");
        String email = readText("New email: ");

        UpdateResult result = students.updateOne(
                studentFilter(name),
                new Document("$set", new Document("email", email))
        );
        printUpdateResult(result);
    }

    // 3. Add a skill to the student's skills array, without duplicates
    private static void addSkill() {
        String name = readText("Student name: ");
        String skill = readText("Skill to add: ");

        UpdateResult result = students.updateOne(
                studentFilter(name),
                new Document("$addToSet", new Document("skills", skill))
        );
        printUpdateResult(result);
    }

    // 4. Update Phone
    private static void updatePhone() {
        String name = readText("Student name: ");
        String phone = readText("New phone number: ");

        UpdateResult result = students.updateOne(
                studentFilter(name),
                new Document("$set", new Document("phone", phone))
        );
        printUpdateResult(result);
    }

    // 5. Update an existing student, or insert one if no name matches
    private static void upsertStudent() {
        String name = readText("Student name: ");
        if (name.isBlank()) {
            System.out.println("Student name cannot be empty.");
            return;
        }

        String email = readText("Email: ");
        String phone = readText("Phone: ");
        String department = readText("Department: ");
        double cgpa = readDouble("CGPA (0-10): ");
        if (!validCgpa(cgpa)) return;

        Document update = new Document("$set",
                new Document("email", email)
                        .append("phone", phone)
                        .append("department", department)
                        .append("cgpa", cgpa))
                .append("$setOnInsert", new Document("name", name));

        UpdateResult result = students.updateOne(
                studentFilter(name),
                update,
                new UpdateOptions().upsert(true)
        );

        if (result.getUpsertedId() != null) {
            System.out.println("Student did not exist; inserted new record.");
        } else if (result.getMatchedCount() > 0) {
            System.out.println("Existing student updated.");
            System.out.println("Modified: " + result.getModifiedCount());
        } else {
            System.out.println("No change made.");
        }
    }

    // 6. Delete one matching student
    private static void deleteStudent() {
        String name = readText("Student name to delete: ");
        String confirm = readText("Type DELETE to confirm: ");

        if (!"DELETE".equals(confirm)) {
            System.out.println("Deletion cancelled.");
            return;
        }

        DeleteResult result = students.deleteOne(studentFilter(name));
        System.out.println("Documents deleted: " + result.getDeletedCount());
    }

    // 7. Delete all students matching a department
    private static void deleteMultiple() {
        String department = readText("Department to delete: ");
        System.out.println("WARNING: This deletes every student in that department.");
        String confirm = readText("Type DELETE " + department + " to confirm: ");

        if (!confirm.equals("DELETE " + department)) {
            System.out.println("Deletion cancelled.");
            return;
        }

        DeleteResult result = students.deleteMany(
                new Document("department", department)
        );
        System.out.println("Students deleted: " + result.getDeletedCount());
    }

    // 8. Count students in each department
    private static void departmentStatistics() {
        System.out.println("\nDepartment Statistics:");

        for (Document doc : students.aggregate(Arrays.asList(
                Aggregates.match(Filters.exists("department")),
                Aggregates.group("$department", Accumulators.sum("count", 1)),
                Aggregates.sort(Sorts.ascending("_id"))
        ))) {
            System.out.println("Department: " + doc.get("_id")
                    + ", Students: " + doc.get("count"));
        }
    }

    // 9. Calculate average CGPA across documents with numeric CGPA values
    private static void averageCGPA() {
        Document result = students.aggregate(Arrays.asList(
                Aggregates.match(new Document("cgpa", new Document("$type", "number"))),
                Aggregates.group(null,
                        Accumulators.avg("averageCGPA", "$cgpa"))
        )).first();

        if (result == null || result.get("averageCGPA") == null) {
            System.out.println("No numeric CGPA records found.");
        } else {
            System.out.println("Average CGPA: " + result.get("averageCGPA"));
        }
    }

    // 10. Show the five highest CGPA records
    private static void topStudents() {
        System.out.println("\nTop 5 Students:");

        for (Document doc : students.aggregate(Arrays.asList(
                Aggregates.match(new Document("cgpa", new Document("$type", "number"))),
                Aggregates.sort(Sorts.descending("cgpa")),
                Aggregates.limit(5),
                Aggregates.project(new Document("name", 1)
                        .append("department", 1)
                        .append("cgpa", 1))
        ))) {
            System.out.println(doc.toJson());
        }
    }

    // 11. Display all students
    private static void viewStudents() {
        boolean found = false;

        for (Document student : students.find()) {
            System.out.println(student.toJson());
            found = true;
        }

        if (!found) {
            System.out.println("No students found.");
        }
    }

    private static void printUpdateResult(UpdateResult result) {
        if (result.getMatchedCount() == 0) {
            System.out.println("Student not found.");
        } else {
            System.out.println("Matched: " + result.getMatchedCount());
            System.out.println("Modified: " + result.getModifiedCount());
        }
    }
}
